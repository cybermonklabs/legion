**`capped-tree.sh`**

````
#!/usr/bin/env bash
# capped-tree.sh <limit-MiB> <seconds> <log> -- <command...>
# capped.sh's interface, but it counts and kills the whole process tree under <command>, found by
# parent pid, so a descendant that starts a session or a process group of its own (Playwright's
# browsers, the e2e harness's servers) is counted and killed too.
set -u
limit_mib=$1 secs=$2 log=$3; shift 4
setsid timeout -s KILL "$secs" "$@" >"$log" 2>&1 &
leader=$!
tree() {
  ps -e -o pid=,ppid=,rss= | awk -v root="$leader" '
    { parent[$1] = $2; rss[$1] = $3 }
    END {
      inside[root] = 1
      do { grown = 0; for (p in parent) if (!(p in inside) && (parent[p] in inside)) { inside[p] = 1; grown = 1 } } while (grown)
      for (p in inside) if (p in rss) print p, rss[p]
    }'
}
peak=0
while kill -0 "$leader" 2>/dev/null; do
  rows=$(tree)
  rss_mib=$(( $(printf '%s\n' "$rows" | awk '{s+=$2} END {print s+0}') / 1024 ))
  [ "$rss_mib" -gt "$peak" ] && peak=$rss_mib
  if [ "$rss_mib" -gt "$limit_mib" ]; then
    kill -KILL $(printf '%s\n' "$rows" | awk '{print $1}') 2>/dev/null
    echo "watchdog: killed tree of $leader at ${rss_mib} MiB resident (limit ${limit_mib} MiB)"
    wait "$leader" 2>/dev/null
    exit 137
  fi
  sleep 1
done
wait "$leader"; status=$?
echo "watchdog: exit=$status peak=${peak} MiB resident (limit ${limit_mib} MiB)"
exit "$status"
````

**`sites-r45.py`**

````
#!/usr/bin/env python3
"""Per-site control of INFRA-1503's stopped-write sentences, in the scratch copy only.

For each call of stoppedWriteNote / stoppedCreate / stoppedCorrection in the web source (not in a
test file, and not the call inside a helper's own definition), the call's value is replaced by the
old failure sentence `Could not save it.` in the scratch copy alone, the tests of that feature
directory run, and the file is restored. A site whose break fails no test of its directory is
reported uncaught; no run is wider than one directory, under the pod's memory rules. Prints one line per site.
"""
import glob, json, os, re, shutil, subprocess, sys

ROOT = "/legion/scratch/tmp/ctl-sites45.xlBH/packages/dispatch"
WS = os.environ["LEGION_WORKSPACE"] + "/packages/dispatch"
WEB = "web/src"
CALL = re.compile(r"\b(stoppedWriteNote|stoppedNote|stoppedCreate|stoppedCorrection)\(")
DEFN = re.compile(r"export function (stoppedWriteNote|stoppedNote|stoppedCreate|stoppedCorrection)\b")


def sites():
    found = []
    for path in sorted(glob.glob(f"{ROOT}/{WEB}/**/*.ts*", recursive=True)):
        rel = os.path.relpath(path, ROOT)
        if ".test." in rel or "__tests__" in rel or rel.endswith("keyed-write.ts"):
            continue
        text = open(path).read()
        helper_bodies = []
        for d in DEFN.finditer(text):
            end = text.find("\n}\n", d.end())
            helper_bodies.append((d.start(), end))
        for m in CALL.finditer(text):
            if any(a <= m.start() <= b for a, b in helper_bodies):
                continue
            if text[max(0, m.start() - 9):m.start()].endswith("function "):
                continue
            line = text.count("\n", 0, m.start()) + 1
            found.append((rel, m.start(), line, m.group(1)))
    return found


def run(target, timeout=900):
    # Under verification.md v48: one Bun process (no --parallel), its whole process tree capped at
    # 2 GiB. A directory the cap kills runs again file by file, stopping at the first file that
    # fails, so a kill is never read as a site no test catches.
    log = "/legion/scratch/tmp/sites-r45-run.log"
    def once(paths):
        out = subprocess.run(["bash", "/legion/scratch/tmp/capped-tree.sh", "2048", str(timeout), log,
                              "--", "bun", "test", "--timeout", "60000",
                              "--preload", "./web/src/__tests__/register-dom.ts",
                              "--preload", "./web/src/__tests__/setup.ts", *paths],
                             cwd=ROOT, capture_output=True, text=True)
        text = open(log, errors="replace").read()
        killed = "watchdog: killed" in out.stdout
        fails = re.findall(r"^\(fail\) (.*?) \[", text, re.M)
        passed = re.search(r"^\s*(\d+) pass$", text, re.M)
        return fails, int(passed.group(1)) if passed else None, killed
    fails, passed, killed = once(target)
    if not killed:
        return fails, passed, 0
    files = sorted(glob.glob(f"{ROOT}/{target[0]}/**/*.test.ts*", recursive=True))
    total = 0
    for f in files:
        fails, passed, killed = once(["./" + os.path.relpath(f, ROOT)])
        if killed:
            print(json.dumps({"killed_file": os.path.relpath(f, ROOT)}), flush=True)
        total += passed or 0
        if fails:
            return fails, total, 0
    return [], total, 0


def main():
    rows = []
    for rel, offset, line, name in sites():
        path = os.path.join(ROOT, rel)
        original = open(path).read()
        broken = original[:offset] + '"Could not save it." ?? ' + original[offset:]
        os.remove(path)
        open(path, "w").write(broken)
        try:
            feature_dir = os.path.dirname(rel)
            fails, passed, code = run([feature_dir])
            scope = feature_dir
        finally:
            os.remove(path)
            shutil.copy2(os.path.join(WS, rel), path)
        row = {"site": f"{rel}:{line}", "call": name, "scope": scope, "failing": len(fails),
               "first": fails[0] if fails else None, "passed": passed}
        rows.append(row)
        print(json.dumps(row), flush=True)
    json.dump(rows, open("/legion/scratch/tmp/sites-r45.json", "w"), indent=1)
    print(f"sites {len(rows)}; caught {sum(1 for r in rows if r['failing'])}; "
          f"uncaught {[r['site'] for r in rows if not r['failing']]}")


main()
````
